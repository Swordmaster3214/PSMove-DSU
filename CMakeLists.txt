cmake_minimum_required(VERSION 3.18)

# Link the MSVC C/C++ runtime statically so the .exe does not require the
# Visual C++ Redistributable on the user's machine. Must be set before project().
option(DSU_STATIC_RUNTIME "Statically link the MSVC runtime (Windows/MSVC only)" ON)
if(DSU_STATIC_RUNTIME)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

# Without this, macOS binaries require the same macOS version as the machine
# that built them (CI runners are always recent). 10.15 is the first release
# with std::filesystem; arm64 builds are raised to 11.0 automatically.
# Must be set before project().
if(CMAKE_HOST_APPLE AND NOT CMAKE_OSX_DEPLOYMENT_TARGET)  # APPLE isn't set until project()
    set(CMAKE_OSX_DEPLOYMENT_TARGET "10.15" CACHE STRING "Minimum macOS version")
endif()

project(dsu_server_psmove LANGUAGES C CXX)
if(APPLE)
    # OBJCXX for PSMoveAPI's .mm files. OBJC is needed too: without it CMake
    # compiles GLFW's .m files as Objective-C++, which rejects GLFW's C code.
    enable_language(OBJC OBJCXX)
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Determine platform suffix for executable name
if(WIN32)
    if(CMAKE_SIZEOF_VOID_P EQUAL 8)
        set(PLATFORM_SUFFIX "win64")
    else()
        set(PLATFORM_SUFFIX "win32")
    endif()
elseif(APPLE)
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
        set(PLATFORM_SUFFIX "macos_arm64")
    else()
        set(PLATFORM_SUFFIX "macos_x64")
    endif()
else()
    # Linux and other Unix-like systems
    if(CMAKE_SIZEOF_VOID_P EQUAL 8)
        if(CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
            set(PLATFORM_SUFFIX "linux_arm64")
        else()
            set(PLATFORM_SUFFIX "linux_x64")
        endif()
    else()
        set(PLATFORM_SUFFIX "linux_x86")
    endif()
endif()

# Set the executable name with platform suffix
set(EXECUTABLE_NAME "dsu_server_psmove_${PLATFORM_SUFFIX}")

# ---------------------------------------------------------------------------
# PSMoveAPI (built from source and linked statically)
#
# PSMoveAPI 4.0.12 only builds a shared library and every prebuilt release is
# shared as well, so we download the pinned sources (plus the hidapi revision
# its submodule points at) and compile only what the server needs straight
# into a static library. The result: no psmoveapi.dll / libpsmoveapi.so has to
# be installed next to (or on the system of) the end user.
#
# To build offline, point CMake at local source trees instead:
#   -DFETCHCONTENT_SOURCE_DIR_PSMOVEAPI=/path/to/psmoveapi-4.0.12
#   -DFETCHCONTENT_SOURCE_DIR_HIDAPI=/path/to/hidapi   (thp/hidapi @ 38681c7)
# ---------------------------------------------------------------------------
include(FetchContent)
if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW) # use extraction time as archive timestamps
endif()

FetchContent_Declare(psmoveapi
    URL      https://github.com/thp/psmoveapi/archive/refs/tags/4.0.12.tar.gz
    URL_HASH SHA256=e4e825c3b799f7ac1d73f0e59de6a2a16bca5e684305fda745d9079401eed0c0
    # psmoveapi has its own CMakeLists.txt; we only want its sources.
    SOURCE_SUBDIR cmake_do_not_add_subdirectory
)
FetchContent_Declare(hidapi
    URL      https://github.com/thp/hidapi/archive/38681c734965636c9fd710dc3898d90aaf923bb9.tar.gz
    URL_HASH SHA256=c50f19fca0e784f26dfae70d908017bb0335d76f80fd288973f0cd1084b96ba2
    SOURCE_SUBDIR cmake_do_not_add_subdirectory
)
FetchContent_MakeAvailable(psmoveapi hidapi)

set(PSMOVEAPI_DIR "${psmoveapi_SOURCE_DIR}")
set(HIDAPI_DIR    "${hidapi_SOURCE_DIR}")

if(NOT EXISTS "${PSMOVEAPI_DIR}/include/psmove.h")
    message(FATAL_ERROR "psmoveapi sources not found at ${PSMOVEAPI_DIR}")
endif()
if(NOT EXISTS "${HIDAPI_DIR}/hidapi/hidapi.h")
    message(FATAL_ERROR "hidapi sources not found at ${HIDAPI_DIR}")
endif()

# psmove_config.h is generated by psmoveapi's own build; reproduce it here.
set(PSMOVEAPI_VERSION_MAJOR 4)
set(PSMOVEAPI_VERSION_MINOR 0)
set(PSMOVEAPI_VERSION_PATCH 12)
set(PSMOVEAPI_GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/psmoveapi_generated")
configure_file(
    "${PSMOVEAPI_DIR}/include/psmove_config.h.in"
    "${PSMOVEAPI_GENERATED_DIR}/psmove_config.h"
    @ONLY
)

# Sources common to every platform
set(PSMOVEAPI_SOURCES
    ${PSMOVEAPI_DIR}/src/psmove.c
    ${PSMOVEAPI_DIR}/src/psmove_calibration.c
    ${PSMOVEAPI_DIR}/src/psmove_orientation.cpp
    ${PSMOVEAPI_DIR}/src/psmoveapi.cpp
    ${PSMOVEAPI_DIR}/src/math/psmove_alignment.cpp
    ${PSMOVEAPI_DIR}/src/math/psmove_math.c
    ${PSMOVEAPI_DIR}/src/math/psmove_quaternion.cpp
    ${PSMOVEAPI_DIR}/src/math/psmove_vector.c
    ${PSMOVEAPI_DIR}/src/daemon/moved_client.cpp
)

# Platform-specific sources (hidapi backend + psmoveapi platform port)
if(WIN32)
    list(APPEND PSMOVEAPI_SOURCES
        ${HIDAPI_DIR}/windows/hid.c
        ${PSMOVEAPI_DIR}/src/platform/psmove_port_windows.c
    )
elseif(APPLE)
    list(APPEND PSMOVEAPI_SOURCES
        ${HIDAPI_DIR}/mac/hid.c
        ${PSMOVEAPI_DIR}/src/platform/psmove_port_osx.mm
        ${PSMOVEAPI_DIR}/src/daemon/moved_monitor_osx.mm
    )
else()
    list(APPEND PSMOVEAPI_SOURCES
        ${HIDAPI_DIR}/linux/hid.c
        ${PSMOVEAPI_DIR}/src/platform/psmove_port_linux.cpp
        ${PSMOVEAPI_DIR}/src/daemon/moved_monitor_linux.c
    )
endif()

add_library(psmoveapi_static STATIC ${PSMOVEAPI_SOURCES})

target_include_directories(psmoveapi_static
    PUBLIC
        ${PSMOVEAPI_GENERATED_DIR}      # psmove_config.h
        ${PSMOVEAPI_DIR}/include        # psmove.h
    PRIVATE
        ${PSMOVEAPI_DIR}/src
        ${PSMOVEAPI_DIR}/src/math
        ${PSMOVEAPI_DIR}/external/glm
        ${HIDAPI_DIR}/hidapi
)

set_target_properties(psmoveapi_static PROPERTIES
    C_STANDARD 11
    C_STANDARD_REQUIRED ON
    CXX_STANDARD 17
    POSITION_INDEPENDENT_CODE ON
)

# psmove.h marks symbols __declspec(dllimport) on Windows unless this is
# defined. We are linking statically, so there is no DLL to import from.
# PUBLIC so that the server target (which includes psmove.h) sees it too.
target_compile_definitions(psmoveapi_static PUBLIC BUILDING_SHARED_LIBRARY)

# This is third-party code; don't drown our build output in its warnings.
if(MSVC)
    target_compile_options(psmoveapi_static PRIVATE /w)
    target_compile_definitions(psmoveapi_static PRIVATE
        _USE_MATH_DEFINES NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS
    )
else()
    target_compile_options(psmoveapi_static PRIVATE -w)
endif()

# Platform dependencies of psmoveapi/hidapi
if(WIN32)
    target_link_libraries(psmoveapi_static PUBLIC
        setupapi bthprops kernel32 ws2_32 winmm version imm32
    )
elseif(APPLE)
    target_link_libraries(psmoveapi_static PUBLIC
        "-framework Foundation"
        "-framework AVFoundation"
        "-framework IOKit"
        "-framework CoreFoundation"
        "-framework IOBluetooth"
    )
else()
    target_compile_definitions(psmoveapi_static PRIVATE _GNU_SOURCE)

    find_package(PkgConfig REQUIRED)
    find_package(Threads REQUIRED)
    pkg_check_modules(DBUS   REQUIRED IMPORTED_TARGET dbus-1)
    pkg_check_modules(UDEV   REQUIRED IMPORTED_TARGET libudev)
    pkg_check_modules(BLUEZ  REQUIRED IMPORTED_TARGET bluez>=5)

    target_link_libraries(psmoveapi_static PUBLIC
        PkgConfig::DBUS PkgConfig::UDEV PkgConfig::BLUEZ Threads::Threads m
    )
endif()

# ---------------------------------------------------------------------------
# GUI: Dear ImGui on GLFW + OpenGL 3 (built from source, linked statically)
#
# Chosen because it is small, has no runtime to ship, and runs the same on
# Windows, macOS and Linux (X11 and Wayland). Turn off with -DDSU_BUILD_GUI=OFF
# for a console-only build; the executable then behaves like --headless.
#
# Offline builds: -DFETCHCONTENT_SOURCE_DIR_IMGUI=... -DFETCHCONTENT_SOURCE_DIR_GLFW=...
# ---------------------------------------------------------------------------
option(DSU_BUILD_GUI "Build the graphical interface" ON)

if(DSU_BUILD_GUI)
    FetchContent_Declare(imgui
        URL      https://github.com/ocornut/imgui/archive/refs/tags/v1.91.9b.tar.gz
        URL_HASH SHA256=8e1bbc76c71d74fef2fb85db7e7ca8eba13d6a86623c54992b60162db554ffdb
        SOURCE_SUBDIR cmake_do_not_add_subdirectory
    )
    FetchContent_Declare(glfw
        URL      https://github.com/glfw/glfw/archive/refs/tags/3.4.tar.gz
        URL_HASH SHA256=c038d34200234d071fae9345bc455e4a8f2f544ab60150765d7704e08f3dac01
    )

    set(GLFW_BUILD_DOCS     OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_TESTS    OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(GLFW_INSTALL        OFF CACHE BOOL "" FORCE)
    set(BUILD_SHARED_LIBS   OFF)  # GLFW as a static library

    if(UNIX AND NOT APPLE)
        # GLFW loads libX11 / libwayland at runtime, so the binary still starts on
        # machines without them (and falls back to headless). Wayland support is
        # compiled in only when its development files are installed.
        find_package(PkgConfig REQUIRED)
        pkg_check_modules(GLFW_WAYLAND_DEPS QUIET wayland-client wayland-cursor wayland-egl xkbcommon)
        find_program(WAYLAND_SCANNER wayland-scanner)
        if(GLFW_WAYLAND_DEPS_FOUND AND WAYLAND_SCANNER)
            set(GLFW_BUILD_WAYLAND ON CACHE BOOL "" FORCE)
            message(STATUS "GUI: building GLFW with X11 and Wayland support")
        else()
            set(GLFW_BUILD_WAYLAND OFF CACHE BOOL "" FORCE)
            message(STATUS "GUI: building GLFW with X11 support only (Wayland dev packages not found)")
        endif()
        set(GLFW_BUILD_X11 ON CACHE BOOL "" FORCE)
    endif()

    FetchContent_MakeAvailable(imgui glfw)
    set(IMGUI_DIR "${imgui_SOURCE_DIR}")

    add_library(imgui_static STATIC
        ${IMGUI_DIR}/imgui.cpp
        ${IMGUI_DIR}/imgui_draw.cpp
        ${IMGUI_DIR}/imgui_tables.cpp
        ${IMGUI_DIR}/imgui_widgets.cpp
        ${IMGUI_DIR}/backends/imgui_impl_glfw.cpp
        ${IMGUI_DIR}/backends/imgui_impl_opengl3.cpp
    )
    target_include_directories(imgui_static PUBLIC ${IMGUI_DIR} ${IMGUI_DIR}/backends)
    target_link_libraries(imgui_static PUBLIC glfw ${CMAKE_DL_LIBS})
    # Stop glfw3.h from including <GL/gl.h>: the GLFW backend makes no GL calls
    # and the OpenGL backend has its own loader, so no GL dev headers are needed.
    target_compile_definitions(imgui_static PRIVATE GLFW_INCLUDE_NONE)
    # ImGui's OpenGL backend loads GL functions itself; only macOS needs the framework linked.
    if(APPLE)
        target_link_libraries(imgui_static PUBLIC "-framework OpenGL")
    elseif(WIN32)
        target_link_libraries(imgui_static PUBLIC opengl32)
    endif()
    set_target_properties(imgui_static PROPERTIES CXX_STANDARD 17 POSITION_INDEPENDENT_CODE ON)
endif()

# Version string shown in the GUI and diagnostics report
find_package(Git QUIET)
set(DSU_VERSION "dev")
if(GIT_FOUND)
    execute_process(
        COMMAND ${GIT_EXECUTABLE} describe --tags --always --dirty
        WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
        OUTPUT_VARIABLE DSU_GIT_DESCRIBE
        OUTPUT_STRIP_TRAILING_WHITESPACE
        ERROR_QUIET
    )
    if(DSU_GIT_DESCRIBE)
        set(DSU_VERSION "${DSU_GIT_DESCRIBE}")
    endif()
endif()

    )

# Create executable with platform-specific name
add_executable(${EXECUTABLE_NAME} ${SOURCES})

target_link_libraries(${EXECUTABLE_NAME} PRIVATE psmoveapi_static)
target_compile_definitions(${EXECUTABLE_NAME} PRIVATE DSU_VERSION="${DSU_VERSION}")

if(DSU_BUILD_GUI)
    target_link_libraries(${EXECUTABLE_NAME} PRIVATE imgui_static)
    target_compile_definitions(${EXECUTABLE_NAME} PRIVATE DSU_WITH_GUI)
endif()

# Platform-specific configurations
if(WIN32)
    # Windows-specific compile definitions to prevent winsock conflicts
    target_compile_definitions(${EXECUTABLE_NAME} PRIVATE
        WIN32_LEAN_AND_MEAN
        NOMINMAX
        _WINSOCKAPI_
    )
elseif(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    # Don't depend on the build machine's libgcc/libstdc++ versions.
    # (System libs such as libudev/libdbus/libbluetooth stay dynamic.)
    target_link_options(${EXECUTABLE_NAME} PRIVATE -static-libgcc -static-libstdc++)
endif()

# Set output directory
set_target_properties(${EXECUTABLE_NAME} PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}
    RUNTIME_OUTPUT_DIRECTORY_DEBUG ${CMAKE_BINARY_DIR}
    RUNTIME_OUTPUT_DIRECTORY_RELEASE ${CMAKE_BINARY_DIR}
)

# Enable debug info for release builds
if(CMAKE_BUILD_TYPE STREQUAL "Release")
    if(WIN32)
        target_compile_options(${EXECUTABLE_NAME} PRIVATE /Zi)
        target_link_options(${EXECUTABLE_NAME} PRIVATE /DEBUG)
    else()
        target_compile_options(${EXECUTABLE_NAME} PRIVATE -g)
    endif()
endif()

# Print the final executable name for confirmation
message(STATUS "Building executable: ${EXECUTABLE_NAME}")
